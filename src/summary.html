overview <- views %>%
  summarize(
    videos = n(),
    creators = n_distinct(creator_id),

    avg_impressions = mean(impressions_n),

Creator Performance

creator_summary <- views %>%
  group_by(creator_id) %>%
  summarise(
    videos_n = n(),
    impressions_total = sum(impressions_n, na.rm = TRUE),
    watch_rate_avg = mean(watch_rate, na.rm = TRUE)
  ) %>%
  arrange(desc(impressions_total))
